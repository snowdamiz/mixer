def require(value, message):
    if not value:
        raise RuntimeError(message)

def canonical(value):
    p = PurePosixPath(value)
    require(str(p) == value and not p.is_absolute() and '..' not in p.parts
            and '.git' not in p.parts, 'Unsafe path: ' + value)
    return p

def read(value):
    canonical(value)
    p = ROOT / value
    for ancestor in [*reversed(p.parents), p]:
        require(not ancestor.is_symlink(), 'Symlink: ' + str(ancestor))
    require(stat.S_ISREG(p.stat().st_mode), 'Nonregular input: ' + value)
    return p.read_bytes()

def digest(data):
    return hashlib.sha256(data).hexdigest()

def info(value, expected=None):
    data = read(value)
    row = {'path': value, 'bytes': len(data), 'sha256': digest(data)}
    if expected is not None:
        require(row['sha256'] == expected['sha256'], 'SHA differs: ' + value)
        require('bytes' not in expected or row['bytes'] == expected['bytes'],
                'Length differs: ' + value)
    return row

def load(value):
    return json.loads(read(value))

def write(name, value):
    p = OUT / name
    require(not p.exists(), 'Refuse overwriting proposal artifact: ' + name)
    p.write_text(json.dumps(value, sort_keys=True, indent=2) + '\n')

def tree(rows):
    return digest(json.dumps(rows, sort_keys=True, separators=(',', ':')).encode())
