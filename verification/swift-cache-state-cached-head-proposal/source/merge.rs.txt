#[repr(C)]
struct CachedMergeCursor<'a> {
    indices: &'a [u8],
    timestamps: &'a [[u8; 8]],
    lp: usize, le: usize, rp: usize, re: usize,
    li: u16, ri: u16, lt: u64, rt: u64,
}
fn cached_load_heads(s: &mut CachedMergeCursor<'_>) {
    let index = s.indices[s.lp];
    s.li = index as u16;
    if index != u8::MAX {
        s.lt = u64::from_le_bytes(s.timestamps[index as usize]);
    }
    let index = s.indices[s.rp];
    s.ri = index as u16;
    if index != u8::MAX {
        s.rt = u64::from_le_bytes(s.timestamps[index as usize]);
    }
}
fn cached_emit(s: &mut CachedMergeCursor<'_>) -> u8 {
    #[cfg(test)]
    {
        if s.li != 256 && s.ri != 256 {
            SORT_COMPARISONS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
        }
        SORT_MOVES.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
    }
    //255 is a real tail byte;256 is private exhausted-cursor control.
    let take_left = if s.ri >= 255 {
        s.li != 256
    } else {
        s.li < 255 && s.lt <= s.rt
    };
    if take_left {
        let out = s.li as u8;
        s.lp += 1;
        if s.lp < s.le {
            let index = s.indices[s.lp];
            s.li = index as u16;
            if index != u8::MAX {
                s.lt = u64::from_le_bytes(s.timestamps[index as usize]);
            }
        } else {
            s.li = 256;
        }
        out
    } else {
        let out = s.ri as u8;
        s.rp += 1;
        if s.rp < s.re {
            let index = s.indices[s.rp];
            s.ri = index as u16;
            if index != u8::MAX {
                s.rt = u64::from_le_bytes(s.timestamps[index as usize]);
            }
        } else {
            s.ri = 256;
        }
        out
    }
}
